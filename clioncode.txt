
#include <iostream>

string imageName;

//first filter
void greyscale(Image& image) {

    for (int i = 0; i < image.width; ++i) {
        for (int j = 0; j < image.height; ++j) {
            unsigned  int avg = 0;
            for (int k = 0; k < 3; ++k) {
                avg += image(i, j, k);
            }

            avg /= 3;
            for (int k = 0; k < 3; ++k) {
                image(i,j,0) = avg;
                image(i,j,1) = avg;
                image(i,j,2) = avg;
            }
        }
    }
    cout << "Image converted to grayscale!" << endl;
}

//fifth filter
void flip(Image& image){
    char choice;
    cout << "Choose flip type (h for horizontal, v for vertical): " << endl;
    cin >> choice;

    int width = image.width;
    int height = image.height;
    if (choice == 'h' || choice == 'H') {
        for (int y = 0; y < height; ++y) {
            for (int x = 0; x < width / 2; ++x) {
                for (int k = 0; k < 3; ++k){
                    swap(image(x, y, k), image(width - 1 - x, y, k));
                }
            }
        }
        cout << "Image flipped horizontally!" << endl;
    }
    else if (choice == 'v' || choice == 'V') {
        for (int y = 0; y < height / 2; ++y) {
            for (int x = 0; x < width; ++x) {
                for (int k = 0; k < 3; ++k) {
                    swap(image(x, y, k), image(x, height - 1 - y, k));
                }
            }
        }
        cout << "Image flipped vertically!" << endl;
    }
    else {
        cout << "Invalid choice, Use 'h' or 'v'" << endl;
    }
}
 //second filter
void baw(Image& image) {
    for (int i = 0 ; i < image.width; i++) {
        for (int j = 0 ; j < image.height; j++) {
            int sum = 0;
            for (int k = 0; k < 3; k++) {
                sum += image(i, j, k);
            }
            int a = sum/3;
            int v;
            if (a > 127) {
                v = 255;
            }
            else {
                v = 0;
            }
            for (int k = 0; k < image.channels; k++) {
                image(i, j, k) = v;
            }
        }
    }
}
// sixth filter
void rotate(Image& img) {
    int degree;
    cout << "enter the degree you want";
    if (degree == 90) {
        Image rotating(img.height,img.width);
        for (int i = 0; i < rotating.width; i++) {
            for (int j = 0; j < rotating.height; j++) {
                for (int k = 0; k < rotating.channels; k++) {
                    rotating(j, img.width - 1 - i,k) = img(i,j,k);
                }
            }
        }
        img = rotating;
    }
    else if (degree == 180) {
        Image rotating(img.height,img.width);
        for (int i = 0; i < rotating.width; i++) {
            for (int j = 0; j < rotating.height; j++) {
                for (int k = 0; k < rotating.channels; k++) {
                   rotating(img.width - 1 - i , img.height - 1 - j,k) = img(i,j,k);
                }
            }
        }
        img = rotating;
    }
    else if (degree == 270) {
        Image rotating(img.height,img.width);
        for (int i = 0; i < img.width; i++) {
            for (int j = 0; j < img.height; j++) {
                for (int k = 0; k < img.channels; k++) {
                    rotating(img.height - 1 - j, i,k) = img(i,j,k);
                }
            }
        }
        img = rotating;
    }

}
